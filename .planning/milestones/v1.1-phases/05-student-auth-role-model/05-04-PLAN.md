```typescript
async function signInStudent(page, request) {
  const supabaseUrl = process.env.NEXT_PUBLIC_SUPABASE_URL!
  const serviceRoleKey = process.env.SUPABASE_SERVICE_ROLE_KEY!
  // generate a magic-link action_link for the fixture student
  const resp = await request.post(`${supabaseUrl}/auth/v1/admin/generate_link`, {
    headers: { apikey: serviceRoleKey, Authorization: `Bearer ${serviceRoleKey}`, 'Content-Type': 'application/json' },
    data: { type: 'magiclink', email: STUDENT_EMAIL },
  })
  expect(resp.ok()).toBeTruthy()
  const body = await resp.json()
  const actionLink = body.action_link ?? body.properties?.action_link
  // follow the link → /auth/callback?type=magiclink → /student-dashboard
  await page.goto(actionLink)
  await page.waitForURL(/\/student-dashboard/, { timeout: 15_000 })
}
```
